cmake_minimum_required(VERSION 3.20)
project(GaiaCompanionAchievementsFoundation C CXX)
set(CMAKE_C_STANDARD 99)
find_package(Threads REQUIRED)
add_library(gaia_companion_achievements STATIC observer.c progress.cpp toolkit/companion_achievements.c toolkit/companion_achievement_presentation.c)
target_compile_features(gaia_companion_achievements PRIVATE cxx_std_17)
target_link_libraries(gaia_companion_achievements PRIVATE Threads::Threads)
target_include_directories(gaia_companion_achievements PUBLIC . toolkit)
if(NOT MSVC)
  target_link_libraries(gaia_companion_achievements PUBLIC m)
endif()
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  enable_testing()
  add_executable(gaia_achievement_observer_test tests/observer_test.c)
  target_link_libraries(gaia_achievement_observer_test PRIVATE gaia_companion_achievements)
  add_test(NAME gaia_achievement_observer COMMAND gaia_achievement_observer_test)
  add_executable(gaia_achievement_progress_test tests/progress_test.cpp)
  target_compile_features(gaia_achievement_progress_test PRIVATE cxx_std_17)
  target_link_libraries(gaia_achievement_progress_test PRIVATE gaia_companion_achievements)
  add_test(NAME gaia_achievement_progress COMMAND gaia_achievement_progress_test)
  # audit.c against stubbed game memory (tests/stubs stand in for the runner).
  add_executable(gaia_achievement_audit_test tests/audit_test.c audit.c)
  target_include_directories(gaia_achievement_audit_test PRIVATE tests/stubs)
  target_compile_definitions(gaia_achievement_audit_test PRIVATE IOG_ACHIEVEMENT_STORAGE=1)
  target_link_libraries(gaia_achievement_audit_test PRIVATE gaia_companion_achievements)
  add_test(NAME gaia_achievement_audit COMMAND gaia_achievement_audit_test)
endif()
